cmake_minimum_required(VERSION 3.16)
project(file_transfer VERSION 1.0.0 LANGUAGES CXX)

# Distribute Windows executables without separate compiler runtime DLLs.
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(transfer_core src/net.cpp src/transfer.cpp)
target_compile_features(transfer_core PUBLIC cxx_std_17)
set_target_properties(transfer_core PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(transfer_core PUBLIC src)
if(MSVC)
    target_compile_options(transfer_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(transfer_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(WIN32)
    target_compile_definitions(transfer_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0600)
    target_link_libraries(transfer_core PUBLIC ws2_32 iphlpapi)
else()
    target_compile_definitions(transfer_core PUBLIC _FILE_OFFSET_BITS=64)
endif()

foreach(program sender recver)
    add_executable(${program} src/${program}.cpp)
    target_link_libraries(${program} PRIVATE transfer_core)
    if(MINGW)
        target_link_options(${program} PRIVATE -municode -static)
    endif()
endforeach()
install(TARGETS sender recver RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    set(integration_command ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
        $<TARGET_FILE:sender> $<TARGET_FILE:recver>)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        # Test-only binaries deterministically force short/interrupted socket operations.
        foreach(program sender recver)
            add_executable(${program}_short_io src/${program}.cpp tests/short_io.cpp)
            target_link_libraries(${program}_short_io PRIVATE transfer_core)
            target_link_options(${program}_short_io PRIVATE -Wl,--wrap=send -Wl,--wrap=recv)
        endforeach()
        list(APPEND integration_command $<TARGET_FILE:sender_short_io> $<TARGET_FILE:recver_short_io>)
    endif()
    add_test(NAME transfer_integration COMMAND ${integration_command})
    set_tests_properties(transfer_integration PROPERTIES TIMEOUT 120)
endif()
